\newenvironment{graphpaper}{%
	\vphantom{\hphantom{.}}%
	\vfill
	\begin{center}%
	\begin{tikzpicture}%
}{%
	\end{tikzpicture}%
	\end{center}%
	\vfill
	\vphantom{\hphantom{.}}%
}

% Command version that includes document wrapper
\newcommand{\graph}[2][]{%
	\ifx\relax#1\relax
		% No description provided
	\else
		\descriptionSidebar{#1}%
	\fi
	\begin{document}%
	\pagestyle{empty}%
	\begin{graphpaper}%
		#2%
	\end{graphpaper}%
	\end{document}%
}
