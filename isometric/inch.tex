\graph[1-inch and $\frac{1}{4}$-inch equilateral triangles]{
	
	\def\spacing{1in}
	\def\width{16}
	\def\height{20}
	
	% Clip to page boundaries (centered on origin)
	\pgfmathsetmacro{\halfwidth}{\width/2}
	\pgfmathsetmacro{\halfheight}{\height/2}
	\clip (-\halfwidth cm, -\halfheight cm) rectangle ++ (\width cm, \height cm);
	
	
	% Define lattice vectors \a and \b
	\pgfmathsetmacro{\sqrtthreeover}{sqrt(3)/2}
	\coordinate (a) at (1 * \spacing, 0cm);
	\coordinate (b) at (0.5 * \spacing, \sqrtthreeover * \spacing);
	
	\foreach \i in {-15,-14,...,15}{
		\foreach \j in {-15,-14,...,15}{
			\draw ($\i*(a) + \j*(b)$) -- ++ (a);
			\draw ($\i*(a) + \j*(b)$) -- ++ (b);
			\draw ($\i*(a) + \j*(b)$) -- ++ ($(a) - (b)$);
		}
	}

	\foreach \dx in {1,2,3}{
		\foreach \i in {-8,-7,...,5}{
			\foreach \j in {-8,-7,...,5}{
				\draw[gray, very thin] ($\i*(a) + \dx/4*(a)+\dx/4*(b) + \j*(b)$) -- ++ (a);
				\draw[gray, very thin] ($\i*(a) + \dx/4*(a) + \j*(b)$) -- ++ (a);
				\draw[gray, very thin] ($\i*(a) + \dx/4*(a) + \j*(b)$) -- ++ (b);
				\draw[gray, very thin] ($\i*(a) + \dx/4*(a) + \j*(b)$) -- ++ ($(a) - (b)$);
			}
		}
	}

}
